**筛选入口**：

```bash
# 1) Exa 全网发现（kind=method）
python - <<'EOF'
import json, glob
for f in glob.glob('90-原始/exa_discovery/*.jsonl'):
    for ln in open(f, encoding='utf-8'):
        if not ln.strip():
            continue
        r = json.loads(ln)
        print(f"{len(r.get('body') or '')}\t{r['title'][:70]}\t{r['url']}")
EOF

# 2) 收入/MRR 类高互动帖（正则粗筛，再人工判读）
python - <<'EOF'
import json, glob, re
pat = re.compile(r'\$\s?\d+[kK]?|MRR|ARR|revenue|收入|营收', re.I)
for f in glob.glob('90-原始/*/*.jsonl'):
    for ln in open(f, encoding='utf-8'):
        if not ln.strip():
            continue
        r = json.loads(ln)
        txt = f"{r['title']} {r.get('body') or ''}"
        if pat.search(txt) and len(r.get('comments') or []) >= 3:
            print(f"{r['source_id']}\t{len(r.get('comments') or [])}\t{r['title'][:60]}\t{r['url']}")
EOF
```
